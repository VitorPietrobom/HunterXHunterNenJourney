"""Split the Hunter x Hunter (2011) OST compilation into game music, sorted by mood.

The compilation is one 80-minute MP3. Tracks were found by silence detection (ffmpeg silencedetect, -40 dB, 1.2 s)
and sorted by measured energy: integrated loudness (EBU R128), high-frequency (>2.5 kHz) and low-frequency (<150 Hz)
level. Calm = quiet and soft, for exploring and training; Fight = loud and bright; Boss = the loudest, heaviest tracks.
Stingers and near-silent tracks are left out.

The music is copyrighted, so the output folder (Resources/Music) is ignored by git: each machine makes its own copy.

Usage: python split_soundtrack.py.txt "path/to/Hunter X Hunter 2011 - OST Compilation.mp3"
"""
import os
import subprocess
import sys

SRC = sys.argv[1]
OUT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "Resources", "Music")

# (start s, end s) of each track in the compilation, by mood.
MOODS = {
    "Calm": [(736.96, 852.40), (978.80, 1106.81), (1112.66, 1223.05), (1483.60, 1568.92), (1969.35, 2096.83),
             (2194.46, 2266.37), (3354.52, 3477.88), (3649.36, 3725.72), (3729.62, 3811.68), (3892.54, 3999.54),
             (4448.42, 4618.51), (4624.58, 4832.70)],
    "Fight": [(0.00, 156.03), (160.98, 284.50), (288.46, 372.51), (857.04, 975.27), (1862.70, 1963.05),
              (2572.84, 2711.98), (3041.35, 3149.77), (3484.05, 3556.38), (4088.40, 4278.13)],
    "Boss": [(1362.24, 1475.53), (1671.43, 1855.65), (469.71, 611.85)],
}

for mood, tracks in MOODS.items():
    folder = os.path.join(OUT, mood)
    os.makedirs(folder, exist_ok=True)
    for i, (start, end) in enumerate(tracks):
        path = os.path.join(folder, f"{mood}{i + 1:02d}.ogg")
        # A short fade at each end so loops and crossfades never click.
        fade_out = max(0.0, end - start - 1.5)
        subprocess.run(["ffmpeg", "-y", "-hide_banner", "-loglevel", "error", "-ss", str(start), "-to", str(end), "-i", SRC,
                        "-af", f"afade=t=in:d=0.8,afade=t=out:st={fade_out}:d=1.5", "-c:a", "libvorbis", "-q:a", "4", path],
                       check=True)
        print(path)
